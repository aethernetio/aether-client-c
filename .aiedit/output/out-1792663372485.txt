<p><b>1. Stream-методы</b> — используют <code>csa_login_stream_t *builder_stream</code> для инкрементальной записи, оборачивая вызов через <code>csa_stream_commit_t</code> и <code>csa_authorized_api_remote_t</code>.</p>
<pre>// csa_login_stream_authorized_api_client (строки 11709-11747)
aether_status_t csa_login_stream_authorized_api_client(
    csa_login_stream_t *builder_stream,
    aether_uuid_t uid,
    const csa_client_api_stream_t * stream) {

    if (builder_stream == NULL ||
        builder_stream->length > builder_stream->capacity) {
        return AETHER_ERR_ARGUMENT;
    }

    if (builder_stream->capacity > 0u &&
        builder_stream->data == NULL) {
        return AETHER_ERR_ARGUMENT;
    }

    size_t remaining =
        builder_stream->capacity - builder_stream->length;

    if (remaining == 0u) {
        return AETHER_ERR_OVERFLOW;
    }

    csa_stream_commit_t commit = {
        &builder_stream->length,
        remaining
    };

    csa_authorized_api_remote_t remote = {
        &commit,
        csa_stream_commit,
        builder_stream->data + builder_stream->length,
        remaining
    };

    return csa_authorized_api_client(
        &remote,
        uid,
        stream);
}</pre>
