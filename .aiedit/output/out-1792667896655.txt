aether_status_t csa_server_registration_api_set_return_key(
    csa_server_registration_api_remote_t *api,
    const common_key_ref_t * key) {

    if (api == NULL ||
        api->send == NULL ||
        api->out == NULL) {
        return AETHER_ERR_ARGUMENT;
    }

    aether_data_out_t *out = api->out;
    if (!aether_data_out_write_byte(out, 6u)) {
        return AETHER_ERR_OVERFLOW;
    }
    if (!common_serialize_key(out, key, 0u)) return false;

    return api->send(
        api->send_ctx,
        api->out->data,
        api->out->pos);
}

aether_status_t csa_server_registration_api_recovery(
    csa_server_registration_api_remote_t *api,
    aether_uuid_t uid,
    const common_key_ref_t * oldMasterKey,
    const common_key_ref_t * newMasterKey) {

    if (api == NULL ||
        api->send == NULL ||
        api->out == NULL) {
        return AETHER_ERR_ARGUMENT;
    }

    aether_data_out_t *out = api->out;
    if (!aether_data_out_write_byte(out, 8u)) {
        return AETHER_ERR_OVERFLOW;
    }
    if (!aether_data_out_write_uuid(out, uid)) return false;
    if (!common_serialize_key(out, oldMasterKey, 0u)) return false;
    if (!common_serialize_key(out, newMasterKey, 0u)) return false;

    return api->send(
        api->send_ctx,
        api->out->data,
        api->out->pos);
}
